package v1

import (
	"context"

	"github.com/grafana/grafana-app-sdk/resource"
)

// TestKind2Backend is implemented by developer-provided code which backs TestKind2's
// storage with something other than the SDK's generic, etcd/CRD-backed store. Implementations
// can be adapted into a rest.Storage via apiserver.NewCustomStorage, to serve TestKind2
// through the aggregated API server without it being backed by a CRD.
type TestKind2Backend interface {
	// Get retrieves a single TestKind2 by its identifier.
	Get(ctx context.Context, identifier resource.Identifier) (*TestKind2, error)
	// List retrieves a list of TestKind2 in the given namespace matching the provided options.
	List(ctx context.Context, namespace string, opts resource.ListOptions) (*TestKind2List, error)
	// Create creates a new TestKind2, returning the created object as it is stored.
	Create(ctx context.Context, obj *TestKind2) (*TestKind2, error)
	// Update updates an existing TestKind2, returning the updated object as it is stored.
	Update(ctx context.Context, obj *TestKind2) (*TestKind2, error)
	// Delete deletes an existing TestKind2 by its identifier.
	Delete(ctx context.Context, identifier resource.Identifier) error
	// Watch begins watching for changes to TestKind2 in the given namespace matching the provided options.
	Watch(ctx context.Context, namespace string, opts resource.WatchOptions) (resource.WatchResponse, error)
}
