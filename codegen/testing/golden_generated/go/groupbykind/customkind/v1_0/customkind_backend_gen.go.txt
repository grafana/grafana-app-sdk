package v1_0

import (
	"context"

	"github.com/grafana/grafana-app-sdk/resource"
)

// CustomKindBackend is implemented by developer-provided code which backs CustomKind's
// storage with something other than the SDK's generic, etcd/CRD-backed store. Implementations
// can be adapted into a rest.Storage via apiserver.NewCustomStorage, to serve CustomKind
// through the aggregated API server without it being backed by a CRD.
type CustomKindBackend interface {
	// Get retrieves a single CustomKind by its identifier.
	Get(ctx context.Context, identifier resource.Identifier) (*CustomKind, error)
	// List retrieves a list of CustomKind in the given namespace matching the provided options.
	List(ctx context.Context, namespace string, opts resource.ListOptions) (*CustomKindList, error)
	// Create creates a new CustomKind, returning the created object as it is stored.
	Create(ctx context.Context, obj *CustomKind) (*CustomKind, error)
	// Update updates an existing CustomKind, returning the updated object as it is stored.
	Update(ctx context.Context, obj *CustomKind) (*CustomKind, error)
	// Delete deletes an existing CustomKind by its identifier.
	Delete(ctx context.Context, identifier resource.Identifier) error
	// Watch begins watching for changes to CustomKind in the given namespace matching the provided options.
	Watch(ctx context.Context, namespace string, opts resource.WatchOptions) (resource.WatchResponse, error)
}
